  not for login, not for navigation, not for any other helper.
- NEVER use a hardcoded `cy.wait(<ms>)` as a substitute for waiting on real
  UI readiness. A fixed sleep is either too short (flaky) or wastefully
  too long (every extra second is paid on every single scenario in the
  file, since login/navigation runs per-test). Always wait on a concrete
  signal instead — `.should('be.visible')`, `.should('not.be.disabled')`,
  `.should('exist')`, or a network wait via `cy.intercept()`/`cy.wait('@alias')`
  — targeting the actual element or request that indicates the UI is
  ready for the next action. The only acceptable use of a numeric
  `cy.wait()` is as a LAST-resort, short buffer (a few hundred ms) after
  an already-asserted state, for a UI animation/transition with no
  observable completion signal — never as the primary wait strategy for
  page loads, navigation, or async data fetches.
  (The #QC TEAM TEMPLATES contain fixed `cy.wait(<ms>)` calls; they are verbatim and exempt. This rule applies only to the code YOU write.)

          next(n + 1);
          return;
        }
      }
    });
  };
  next(0);

// The ERP shows error / info popups after opening a module or a screen. Which
// popups appear (and how many) differs per screen, so none can be scripted in
// advance. This closes whatever is visible, repeatedly, until a quiet period
// passes with nothing on screen - so it does nothing on a screen with no popups
// and copes with a screen that shows many. Each popup's text is logged, never
// hidden: a popup on load usually means a failed API call.
function dismissPopups({ quietMs = 1500, max = 15 } = {}) {
  const CLOSERS = [
    '.mat-mdc-dialog-component-host > .mat-icon',   // dialog close icon
    'div.alertbutton',                              // OK / confirm button
  ];
  const next = (n) => {
    if (n >= max) return;
    cy.wait(quietMs);                               // popups arrive late: wait for a quiet moment
    cy.get('body').then(($body) => {
      for (const sel of CLOSERS) {
        const $el = $body.find(sel).filter(':visible');
        if ($el.length) {
          const msg = $body.find('.mat-mdc-dialog-component-host').first().text().replace(/\s+/g, ' ').trim().slice(0, 200);
          Cypress.log({ name: 'popup', message: msg || sel });
          cy.wrap($el.first()).click({ force: true });
          next(n + 1);                              // check again: another popup may have appeared
          return;
        }
      }
    });
  };
  next(0);
}
